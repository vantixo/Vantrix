/**
 * Vantrix Character Seeds — Production Launch Set
 *
 * All characters from the product vision notes, fully realized with:
 *   - Complete backstory and personality
 *   - Human themes (love, betrayal, death, light/dark, religion, etc.)
 *   - Speech style and occupation
 *   - Profession-based characters with real domain knowledge depth
 *   - Interconnected lore (Ghost of Muru ↔ Yanefes relationship)
 *
 * The psychological/emotional internals (attachment, mood, drift) are built
 * from these seeds invisibly — never exposed to the user.
 */

export interface CharacterSeed {
  name:         string;
  age:          number;
  gender:       "female" | "male" | "anime";
  category:     string;
  description:  string;
  personality:  string;
  backstory:    string;
  scenario:     string;
  occupation:   string;
  speech_style: string;
  tags:         string[];
  archetype:    string;
  opening_line: string;
  is_featured:  boolean;
  is_premium:   boolean;
  /**
   * Real per-character tier gate, enforced by checkCharacterTierAccess()
   * (src/lib/rate-limit/index.ts). Defaults to 'free' when omitted. Only set
   * this above 'free' for characters that should require a paid plan —
   * 'elite' is the only truly VIP-exclusive tier; keep that reserved for a
   * small handful of characters (see 20260721_character_tier_separation.sql
   * for the current authoritative assignment, which this field should stay
   * in sync with).
   */
  min_tier?:    "free" | "spark" | "basic" | "premium" | "elite" | "enterprise";
  tokens_cost:  number;
  // Internal prompt enrichment fields
  values_list:  string[];
  fears:        string[];
  dreams:       string[];
  flaws:        string[];
  current_goal: string;
  daily_routine: string[];
  theme:        string; // human theme from notes
  /**
   * Optional. Most seeds omit this — image_url defaults to
   * /images/character-placeholder.png (see 20260624_character_image_url_
   * default.sql) and a real portrait is generated afterward via the
   * Fal.ai LoRA pipeline (POST /api/admin/generate-character-portraits).
   * Set this when a real portrait already exists at seed time (e.g. a
   * commissioned/uploaded image) so the seed migration can write it
   * directly and skip the generate step for that character.
   */
  image_url?:   string;
  /**
   * Optional. A locked-canon LoRA training prompt in the same convention as
   * canon.ts's face_prompt (vtx_<slug>, physical descriptors, NO-change
   * clauses). Not currently consumed by the portrait generation pipeline —
   * seed characters today get their portraits from buildAppearancePrompt()
   * in generate-character-portraits/route.ts, driven by the DB's
   * hair_color/eye_color/body_type/skin_tone columns, not this field.
   * Wiring a seed character onto the locked face_prompt + LoRA-training
   * flow that canon characters use is a real infra/cost decision (each
   * LoRA run has real training time/cost) and should be a deliberate
   * per-character choice, not automatic just because this field is
   * populated — hence it's stored but not yet read anywhere.
   */
  face_prompt?: string;
  /**
   * Optional. Additional display images beyond the primary image_url,
   * shown in the character's gallery/carousel. Maps to the
   * gallery_image_urls TEXT[] column (see
   * 20260717_character_media_gallery.sql). Distinct from reference_images
   * (LoRA training inputs) — these are public-facing.
   */
  gallery_image_urls?: string[];
}
