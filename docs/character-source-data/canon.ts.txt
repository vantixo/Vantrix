// src/lib/characters/canon.ts
// ─────────────────────────────────────────────────────────────────────────────
// Official Vantrix canon characters.
// Every attribute here maps 1:1 to the characters table schema.
// The face_prompt is the locked canonical description for LoRA training.
// NEVER change physical attributes after a LoRA has been trained.
// ─────────────────────────────────────────────────────────────────────────────

export interface CanonCharacter {
  slug:              string;
  name:              string;
  age:               number;
  ethnicity:         string;
  tagline:           string;
  archetype:         string;

  // Physical canon (locked post-training)
  physical: {
    height:          string;
    body_type:       string;
    face_shape:      string;
    eye_color:       string;
    hair_color:      string;
    hair_style:      string;
    skin_tone:       string;
    nose_type:       string;
    lip_type:        string;
    signature_items: string[];
  };

  // Canonical LoRA training prompt
  face_prompt:       string;

  /**
   * Optional. Appended to face_prompt at scene-generation time only (lens,
   * lighting, texture, grain direction) — NEVER merged into face_prompt
   * itself, since that field is LoRA-locked and touching it means
   * retraining. This is purely an image-quality layer for however the
   * generation pipeline chooses to combine it with face_prompt + a
   * MoodRoom's baseScene; it is not yet wired into scene-generator.ts /
   * lora-pipeline.ts — populated here as structured data for that future
   * wiring decision, not currently consumed at runtime.
   */
  generation_style?: string;

  // Personality
  personality:       string;
  backstory:         string;
  origin:            string;
  occupation:        string;
  speech_style:      string;
  opening_line:      string;
  love_language:     string;
  current_goal:      string;

  // Trait scores
  traits: {
    openness:        number;  // 0–100
    warmth:          number;
    adventure:       number;
    depth:           number;
  };

  // Narrative arrays
  values:            string[];
  fears:             string[];
  dreams:            string[];
  flaws:             string[];
  secrets:           string[];

  // Mood rooms this character inhabits
  mood_rooms:        string[];

  // Discovery
  is_nsfw:           boolean;
  min_tier:          'free' | 'spark' | 'basic' | 'premium' | 'elite';
  is_featured:       boolean;
  featured_position: number;

  // Canvas scenes — prompts that change scene while character stays locked
  scene_presets:     ScenePreset[];
}

export interface ScenePreset {
  id:           string;
  label:        string;
  mood_room:    string;
  scene_prompt: string;  // appended to face_prompt for generation
}

export function getCanonCharacterBySlug(slug: string): CanonCharacter | undefined {
  return CANON_CHARACTERS.find(c => c.slug === slug);
}

export function getFeaturedCharacters(): CanonCharacter[] {
  return CANON_CHARACTERS
    .filter(c => c.is_featured)
    .sort((a, b) => a.featured_position - b.featured_position);
}

export function getScenePresetsForCharacter(slug: string): ScenePreset[] {
  return getCanonCharacterBySlug(slug)?.scene_presets ?? [];
}

/** Build the full Fal.ai generation prompt for a specific scene */
export function buildScenePrompt(character: CanonCharacter, sceneOverride?: string): string {
  const baseScene = sceneOverride ?? character.scene_presets[0]?.scene_prompt ?? '';
  return `${character.face_prompt} ${baseScene}`;
}
